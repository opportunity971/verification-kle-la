// ============================================================ // PASSERELLE GOOGLE SHEET <-> PAGE WEB "KLÉ LA MASS / MISISYEN" // ============================================================ var ROWS = 40; var COLS = 5; // Change ce mot de passe si besoin (reste côté serveur, jamais visible du client) var ORGA_PASSWORD = 'orgakléla2026'; function getSheet(name) { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); } // Initialise la feuille "Grille" avec les 200 lignes si elle est vide. // Se lance automatiquement au premier appel, ne fait rien si déjà rempli. function ensureGridInitialized() { var sheet = getSheet('Grille'); if (sheet.getLastRow() >= 1) return; // déjà initialisé sheet.appendRow(['Rangee', 'Position', 'Instrument', 'Nom', 'Bloquee', 'Supprimee']); var rows = []; for (var r = 1; r <= ROWS; r++) { for (var c = 1; c <= COLS; c++) { rows.push([r, c, '', '', false, false]); } } sheet.getRange(2, 1, rows.length, 6).setValues(rows); } function readGrid() { ensureGridInitialized(); var sheet = getSheet('Grille'); var data = sheet.getDataRange().getValues(); var grid = {}; for (var i = 1; i < data.length; i++) { var row = data[i]; var key = row[0] + '-' + row[1]; grid[key] = { instrument: row[2] || '', name: row[3] || '', blocked: row[4] === true, removed: row[5] === true }; } return grid; } function findRowIndexForCell(sheet, r, c) { // Ligne 2 = (r-1)*COLS + c + 1 (ligne 1 = en-têtes), calcul direct car la grille est générée dans cet ordre return (r - 1) * COLS + c + 1; } function writeCell(r, c, instrument, name, blocked, removed) { var sheet = getSheet('Grille'); ensureGridInitialized(); var rowIndex = findRowIndexForCell(sheet, r, c); sheet.getRange(rowIndex, 3, 1, 4).setValues([[instrument, name, blocked, removed]]); } function readRoster() { var sheet = getSheet('Roster'); var data = sheet.getDataRange().getValues(); var names = []; for (var i = 0; i < data.length; i++) { var v = (data[i][0] || '').toString().trim(); if (v) names.push(v); } return names; } function writeRoster(names) { var sheet = getSheet('Roster'); sheet.clear(); if (names.length > 0) { sheet.getRange(1, 1, names.length, 1).setValues(names.map(function(n){ return [n]; })); } } function readLabels() { var sheet = getSheet('Etiquettes'); var data = sheet.getDataRange().getValues(); var labels = {}; for (var i = 0; i < data.length; i++) { var r = data[i][0]; var instrument = data[i][1]; if (r && instrument) labels[r] = instrument; } return labels; } function writeLabels(labels) { var sheet = getSheet('Etiquettes'); sheet.clear(); var rows = []; for (var key in labels) { rows.push([Number(key), labels[key]]); } if (rows.length > 0) { sheet.getRange(1, 1, rows.length, 2).setValues(rows); } } function normalizeName(s) { return (s || '').toString().trim().toLowerCase().replace(/\s+/g, ' '); } // ------------------------------------------------------------ // doGet : lecture (pas de mot de passe nécessaire) // ------------------------------------------------------------ function doGet(e) { var out = { grid: readGrid(), roster: readRoster(), labels: readLabels() }; return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------ // doPost : écriture (verrouillage pour éviter les conflits) // ------------------------------------------------------------ function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var body = JSON.parse(e.postData.contents); var action = body.action; var result = { ok: false }; if (action === 'reserve') { var r = body.row, c = body.col, name = body.name, instrument = body.instrument; var grid = readGrid(); var key = r + '-' + c; var cell = grid[key]; if (!cell || cell.name || cell.blocked || cell.removed) { result = { ok: false, error: 'Place indisponible.' }; } else if (cell.instrument && cell.instrument !== instrument) { result = { ok: false, error: "Cet instrument ne correspond pas à celui de cette rangée." }; } else if (!cell.instrument && !instrument) { result = { ok: false, error: 'Choisis un instrument.' }; } else { var alreadyHasSeat = Object.keys(grid).some(function(k) { return grid[k].name && normalizeName(grid[k].name) === normalizeName(name); }); if (alreadyHasSeat) { result = { ok: false, error: 'Ce musicien a déjà une place.' }; } else { var finalInstrument = cell.instrument || instrument; writeCell(r, c, finalInstrument, name, false, false); result = { ok: true }; } } } else if (action === 'release') { var r = body.row, c = body.col, name = body.name; var grid = readGrid(); var cell = grid[r + '-' + c]; if (cell && normalizeName(cell.name) === normalizeName(name)) { var labels = readLabels(); var keepInstrument = labels[r] ? cell.instrument : ''; writeCell(r, c, keepInstrument, '', false, false); result = { ok: true }; } else { result = { ok: false, error: 'Cette place ne t\'appartient pas.' }; } } else if (body.password !== ORGA_PASSWORD) { result = { ok: false, error: 'Mot de passe organisateur incorrect.' }; } else if (action === 'checkOrgaPassword') { result = { ok: true }; } else if (action === 'setCell') { writeCell(body.row, body.col, body.instrument || '', body.name || '', !!body.blocked, !!body.removed); result = { ok: true }; } else if (action === 'setRowInstrument') { var r = body.row, instrument = body.instrument; var grid = readGrid(); for (var c = 1; c <= COLS; c++) { var cell = grid[r + '-' + c] || { name: '', blocked: false }; writeCell(r, c, instrument, cell.name, cell.blocked, false); } var labels = readLabels(); if (instrument) { labels[r] = instrument; } else { delete labels[r]; } writeLabels(labels); result = { ok: true }; } else if (action === 'resetGrid') { var sheet = getSheet('Grille'); sheet.clear(); ensureGridInitialized(); writeLabels({}); result = { ok: true }; } else if (action === 'addRosterName') { var roster = readRoster(); if (!roster.some(function(n){ return normalizeName(n) === normalizeName(body.name); })) { roster.push(body.name); writeRoster(roster); } result = { ok: true }; } else if (action === 'addRosterBulk') { var roster = readRoster(); var added = 0; (body.names || []).forEach(function(n) { if (!roster.some(function(x){ return normalizeName(x) === normalizeName(n); })) { roster.push(n); added++; } }); writeRoster(roster); result = { ok: true, added: added }; } else if (action === 'removeRosterName') { var roster = readRoster().filter(function(n){ return normalizeName(n) !== normalizeName(body.name); }); writeRoster(roster); result = { ok: true }; } else { result = { ok: false, error: 'Action inconnue.' }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
